package neiro.simple.nlp;

import neiro.simple.mlp.Net;

import java.util.List;

import static neiro.simple.mlp.Net.*;


public class VocToEmbed {

    /**Для словаря слов получить эмбединги путём обучения нейронки и снятия значений с промежуточного слоя нейронки*/
    public static double[][] vocToEmbeding(List<String> vocabulary, int embedingSize){

        int size = 1000; //vocabulary.size()

        Net net = new Net(List.of(
                (layer) -> new LayerInput(size, 0.0),
                (layer) -> new LayerMedium.LayerMediumTanh(layer, size /10, 0.0),
                (layer) -> new LayerOutput.LayerOutputSigmoid(layer,size)
        ));

        net.trains(
                (Integer i) -> generateVec(i, size),
                (Integer i) -> generateVec(i, size),
                size,
                0.2,
                9000,
                10000,
                new Net.TestWrapper(
                        (Integer i) -> generateVec(i, size),
                        (Integer i) -> generateVec(i, size),
                        size,
                        estimationMax,
                        0.1),
                0.95
        );


        //снимаем эмбединги
        double embeding[][] = new double[size][embedingSize];
        for(int i = 0; i < size; i++){
            net.forward(generateVec(i, size));
            for(int j = 0; j<embedingSize; j++)
                embeding[i][j] = net.getLayers().get(1).neirons.get(j).oValue;
        }

        return embeding;
    }


}

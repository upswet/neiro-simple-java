package neiro.simple.rnn;

import java.util.Arrays;
import java.util.function.Function;
import java.util.function.Supplier;
import java.util.function.UnaryOperator;

/**Промежуточный слой для rnn-сетей*/
public class LayerRnnMedium extends LayerMedium {
    double[] previousState;// Состояние памяти для рекуррентных связей

    /**
     * Конструктор промежуточного слоя
     *
     * @param nCount        - количество нейронов в слое
     * @param prevoisLayer  - предыдущий слой
     * @param activation    - фун активации
     * @param activationDer - производная фун активации
     * @param initWeightFun - функция инициализации весов
     * @return - промежуточный слой
     */
    public LayerRnnMedium(int nCount, Layer prevoisLayer, UnaryOperator<Double> activation, Function<Neiron, Double> activationDer, Supplier<Double> initWeightFun) {
        super(nCount, prevoisLayer, activation, activationDer, initWeightFun, 0.0); //классический дропаут не примени к rnn так как они развёрнуты во времени

        previousState = new double[nCount];
        resetState();
    }

    @Override
    public void forward(boolean trainingMode) {
        // Для каждого нейрона вычисляем взвешенную сумму
        int i=0;
        for(Neiron n : neirons) {
            n.iValue=n.bias + previousState[i] * 0.3;// Начинаем с bias и рекуррентной составляющей
            for (Link link : n.iLinks)
                n.iValue  += link.iNeiron.oValue * link.weight;
            n.oValue=this.activation.apply(n.iValue);

            // Сохраняем текущее состояние для следующего шага
            previousState[i] = n.oValue;

            i++;
        }
    }

    @Override
    public void backward(double lr, boolean trainingMode) {
        super.backward(lr, trainingMode);
    }

    // Метод для сброса состояния
    public void resetState() {
        Arrays.fill(previousState, 0.0);
    }
}
